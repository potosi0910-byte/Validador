/* Cada fila (tecnologia) con la prioridad Y el nombre de agrupador de
   SU PROPIO codigo, usando SOLO el catalogo de codigos decisivos. Si el
   codigo no esta (porque nunca clasifica solo, o no aparece en el Excel)
   se le da prioridad 99 y agrupador NULL. */
ConPrioridadCodigo AS (
    SELECT
        pc.*,
        ISNULL(cu.Prioridad, 99)             AS PrioridadCodigo,
        cu.Agrupador                          AS AgrupadorCodigo
    FROM [BODEGAS].[dbo].[ProduccionCostos] pc
    LEFT JOIN ClasificacionCUPS cu
           ON cu.Codigo = pc.[COD_CUPS]
    WHERE
        TRY_CONVERT(DATE, pc.FEC_SERVICIO) >= '2026-06-01'
        AND TRY_CONVERT(DATE, pc.FEC_SERVICIO) < '2026-06-30'
        AND LTRIM(RTRIM(CAST(pc.IDE_EAPBA AS NVARCHAR(100)))) = '900935126'
),

/* La ATENCION se identifica UNICAMENTE por CONS_INGRESO (un ingreso = una
   atencion, sin importar cuantos servicios/CUPS tenga). Un mismo CONS_INGRESO
   solo puede tener UN agrupador, que se repite en todas sus filas.

   El agrupador NO se deriva de una tabla fija numero->nombre (eso rompia
   en cuanto dos codigos distintos comparten el mismo numero de prioridad
   pero tienen nombres diferentes, como 110A01="UCI Adultos" y
   120B01="Cuidados Basicos neonatal", ambos podrian coincidir en prioridad).
   En vez de eso, se toma el NOMBRE REAL del codigo que gano la prioridad
   mas alta (numero mas bajo) dentro del ingreso, con FIRST_VALUE. Ej.: si
   110A01 (UCI Adultos, prioridad 1) esta entre los CUPS del ingreso, ese
   es el codigo que define el agrupador de TODO el ingreso. */
ConPrioridadPrestacion AS (
    SELECT
        *,
        MIN(PrioridadCodigo) OVER (
            PARTITION BY CONS_INGRESO
        ) AS PrioridadPrestacion,
        FIRST_VALUE(ISNULL(AgrupadorCodigo, N'Sin Clasificar')) OVER (
            PARTITION BY CONS_INGRESO
            ORDER BY PrioridadCodigo ASC, COD_CUPS ASC
        ) AS AgrupadorPrestacion
    FROM ConPrioridadCodigo
)
